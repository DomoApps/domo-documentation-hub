export const DeprecatedNote = ({ replacement, replacementLabel = "the recommended replacement" }) => (
  <Warning>
    <strong>Deprecated:</strong>{" "}
    This feature is officially deprecated. It remains functional but is no longer recommended.{" "}
    {replacement
      ? <>Please migrate to <a href={replacement}>{replacementLabel}</a> to avoid disruption when this feature is removed.</>
      : "Please migrate to the recommended replacement to avoid disruption when this feature is removed."
    }
  </Warning>
);
